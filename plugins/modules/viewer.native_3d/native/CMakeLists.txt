cmake_minimum_required(VERSION 3.20)
project(gx3d LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(VTK 9 REQUIRED COMPONENTS CommonCore CommonDataModel FiltersCore FiltersGeneral FiltersGeometry FiltersHybrid FiltersModeling FiltersSources IOGeometry IOImage IOXML RenderingCore RenderingOpenGL2 RenderingAnnotation InteractionStyle InteractionWidgets)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
add_library(gx3d SHARED src/gx3d.cpp)
target_include_directories(gx3d PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(gx3d PRIVATE ${VTK_LIBRARIES})
if(UNIX AND NOT APPLE)
  target_link_libraries(gx3d PRIVATE dl)
endif()
vtk_module_autoinit(TARGETS gx3d MODULES ${VTK_LIBRARIES})
set_target_properties(gx3d PROPERTIES INSTALL_RPATH "$ORIGIN")
option(GX3D_WITH_OCCT "Build native STEP import with Open CASCADE Technology" OFF)
set(GX3D_OCCT_SYSROOT "" CACHE PATH "Optional OCCT development sysroot")
add_executable(gx3d-worker src/worker.cpp)
target_link_libraries(gx3d-worker PRIVATE ${VTK_LIBRARIES} OpenSSL::Crypto)
if(WIN32)
  target_link_libraries(gx3d-worker PRIVATE psapi)
endif()
if(GX3D_WITH_OCCT)
  if(GX3D_OCCT_SYSROOT)
    target_include_directories(gx3d-worker PRIVATE "${GX3D_OCCT_SYSROOT}/usr/include/opencascade")
    set(GX3D_OCCT_LIBDIR "${GX3D_OCCT_SYSROOT}/usr/lib/x86_64-linux-gnu")
    target_link_directories(gx3d-worker PRIVATE "${GX3D_OCCT_LIBDIR}")
    target_link_options(gx3d-worker PRIVATE "-Wl,-rpath-link,${GX3D_OCCT_LIBDIR}")
    target_link_libraries(gx3d-worker PRIVATE TKDESTEP TKXSBase TKXCAF TKCAF TKLCAF TKBRep TKMesh TKGeomBase TKMath TKernel)
  else()
    find_package(OpenCASCADE REQUIRED COMPONENTS FoundationClasses ModelingData ModelingAlgorithms Visualization ApplicationFramework DataExchange)
    target_include_directories(gx3d-worker PRIVATE "${OpenCASCADE_INCLUDE_DIR}")
    target_link_libraries(gx3d-worker PRIVATE TKDESTEP TKXSBase TKXCAF TKCAF TKLCAF TKBRep TKMesh TKMath)
  endif()
  target_compile_definitions(gx3d-worker PRIVATE GX3D_WITH_OCCT=1)
  if(UNIX AND NOT APPLE)
    target_sources(gx3d-worker PRIVATE src/step_transfer_cache.cpp)
  endif()
endif()
vtk_module_autoinit(TARGETS gx3d-worker MODULES ${VTK_LIBRARIES})
set_target_properties(gx3d-worker PROPERTIES INSTALL_RPATH "$ORIGIN/../lib")
add_executable(gx3d-host src/host.cpp)
target_link_libraries(gx3d-host PRIVATE gx3d ${VTK_LIBRARIES} OpenSSL::Crypto)
if(WIN32)
  target_link_libraries(gx3d-host PRIVATE psapi)
endif()
vtk_module_autoinit(TARGETS gx3d-host MODULES ${VTK_LIBRARIES})
set_target_properties(gx3d-host PROPERTIES INSTALL_RPATH "$ORIGIN/../lib")
install(TARGETS gx3d-worker gx3d-host RUNTIME DESTINATION bin)
install(TARGETS gx3d RUNTIME DESTINATION bin LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
install(FILES include/gx3d.h DESTINATION include)
