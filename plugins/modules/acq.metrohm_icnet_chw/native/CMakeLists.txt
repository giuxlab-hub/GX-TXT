# Development/package build only; never invoked by installed runtime.
cmake_minimum_required(VERSION 3.16)
project(gxtxt_chw_worker VERSION 0.3.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_executable(chw_worker chw_reader.c chw_worker.c)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(chw_worker PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(UNIX)
  target_link_libraries(chw_worker PRIVATE m)
endif()
install(TARGETS chw_worker RUNTIME DESTINATION bin)
