cmake_minimum_required(VERSION 3.12)
project(gxtxt_parameters LANGUAGES C)
add_library(gxparameters_core STATIC gx_parameters.c)
target_compile_features(gxparameters_core PUBLIC c_std_99)
target_include_directories(gxparameters_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
set_target_properties(gxparameters_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(NOT MSVC)
  target_link_libraries(gxparameters_core PUBLIC m)
endif()
find_package(TCL REQUIRED)
add_library(gxparameters MODULE gx_parameters_tcl.c)
target_include_directories(gxparameters PRIVATE ${TCL_INCLUDE_PATH})
target_link_libraries(gxparameters PRIVATE gxparameters_core ${TCL_LIBRARY})
set_target_properties(gxparameters PROPERTIES PREFIX "")
