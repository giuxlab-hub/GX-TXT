cmake_minimum_required(VERSION 3.16)
project(gxtxt_stderr_diagnostics VERSION 1.0 LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Diagnostics v1 requires Linux UNIX IPC and safe openat/flock")
endif()
option(GXDIAG_PRIVATE_FIXTURES "Compile private fault injection (never package these artifacts)" OFF)
find_package(TCL REQUIRED)
find_package(Threads REQUIRED)
add_executable(gxtxt-diagnostics-helper helper.cpp)
target_compile_features(gxtxt-diagnostics-helper PRIVATE cxx_std_17)
target_link_libraries(gxtxt-diagnostics-helper PRIVATE Threads::Threads)
add_library(gxdiagnostics MODULE adapter.cpp producer.cpp)
target_compile_features(gxdiagnostics PRIVATE cxx_std_17)
target_include_directories(gxdiagnostics PRIVATE ${TCL_INCLUDE_PATH})
target_link_libraries(gxdiagnostics PRIVATE ${TCL_LIBRARY} Threads::Threads ${CMAKE_DL_LIBS})
set_target_properties(gxdiagnostics PROPERTIES PREFIX "")
foreach(target gxdiagnostics gxtxt-diagnostics-helper)
  if(GXDIAG_PRIVATE_FIXTURES)
    target_compile_definitions(${target} PRIVATE GXDIAG_PRIVATE_FIXTURES=1)
  endif()
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  install(TARGETS ${target} LIBRARY DESTINATION core/diagnostics/bin RUNTIME DESTINATION core/diagnostics/bin)
endforeach()
