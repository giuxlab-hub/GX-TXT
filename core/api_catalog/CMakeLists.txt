cmake_minimum_required(VERSION 3.16)
project(gxtxt_api_catalog VERSION 1.1 LANGUAGES CXX)
find_package(TCL 8.6 REQUIRED)
find_package(SQLite3 REQUIRED)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
add_library(gxtxt_api_catalog MODULE api_catalog_tcl.cpp)
target_compile_features(gxtxt_api_catalog PRIVATE cxx_std_17)
target_include_directories(gxtxt_api_catalog PRIVATE ${TCL_INCLUDE_PATH})
target_link_libraries(gxtxt_api_catalog PRIVATE ${TCL_LIBRARY} SQLite::SQLite3 OpenSSL::Crypto)
set_target_properties(gxtxt_api_catalog PROPERTIES PREFIX "lib")
if(MSVC)
  target_compile_options(gxtxt_api_catalog PRIVATE /W4 /EHsc)
else()
  target_compile_options(gxtxt_api_catalog PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
  target_link_options(gxtxt_api_catalog PRIVATE -Wl,-z,relro,-z,now)
endif()
install(TARGETS gxtxt_api_catalog LIBRARY DESTINATION core/api_catalog RUNTIME DESTINATION core/api_catalog)
